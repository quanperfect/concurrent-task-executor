cmake_minimum_required(VERSION 3.25)
project(concurrent_task_executor VERSION 1.0.0 LANGUAGES CXX)

option(CTE_BUILD_TESTS "Build correctness and stress tests" ${PROJECT_IS_TOP_LEVEL})
option(CTE_BUILD_EXAMPLES "Build public API examples" ${PROJECT_IS_TOP_LEVEL})
option(CTE_BUILD_BENCHMARKS "Build measurement tools" ${PROJECT_IS_TOP_LEVEL})
option(CTE_FETCH_TEST_DEPS "Fetch pinned GoogleTest if a package is unavailable" OFF)
option(CTE_WARNINGS_AS_ERRORS "Treat project warnings as errors" ${PROJECT_IS_TOP_LEVEL})
set(CTE_SANITIZER "none" CACHE STRING "none, thread, or address-undefined")
set_property(CACHE CTE_SANITIZER PROPERTY STRINGS none thread address-undefined)

find_package(Threads REQUIRED)
include(cmake/ProjectOptions.cmake)

add_library(cte STATIC src/task_executor.cpp)
add_library(cte::cte ALIAS cte)
target_include_directories(cte PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>")
target_link_libraries(cte PUBLIC Threads::Threads)
target_compile_features(cte PUBLIC cxx_std_20)
cte_project_options(cte)

if(CTE_BUILD_TESTS)
    enable_testing()
    include(cmake/TestDependencies.cmake)
    add_subdirectory(tests)
endif()
if(CTE_BUILD_EXAMPLES)
    add_subdirectory(examples)
endif()
if(CTE_BUILD_BENCHMARKS)
    add_subdirectory(benchmarks)
endif()
